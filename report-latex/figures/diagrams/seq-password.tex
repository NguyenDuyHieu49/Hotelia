\begin{seqdiagram}
  \participant{app}{0}{Ứng dụng iOS}
  \participant{ctl}{3.2}{Auth-\\Controller}
  \participant[29mm]{svc}{6.6}{PasswordReset\\Service}
  \participant{db}{9.9}{MongoDB}
  \participant{mail}{13.1}{Dịch vụ gửi email}
  \msg{app}{ctl}{POST /auth/forgot-password (email)}
  \msg{ctl}{svc}{request(email)}
  \msg{svc}{db}{tài khoản đang hoạt động? đã gửi trong 60 giây qua?}
  \self{svc}{tạo mã ngẫu nhiên 24 byte, băm SHA-256}
  \msg{svc}{db}{lưu tokenHash, hết hạn sau 15 phút}
  \msg{svc}{mail}{gửi mã qua HTTPS (Bearer, tối đa 10 giây)}
  \fragopen{app}{mail}{alt}{[gửi thất bại]}
  \msg{svc}{db}{xóa mã vừa tạo}
  \ret{svc}{app}{503: dịch vụ email không khả dụng}
  \fragelse{[gửi được, hoặc email không tồn tại]}
  \ret{svc}{app}{thông báo chung, không lộ email có đăng ký hay không}
  \fragclose
  \msg{app}{ctl}{POST /auth/reset-password (code, password)}
  \msg{ctl}{svc}{reset(dto)}
  \seqadvance[0.35]
  \msg{svc}{db}{giao dịch: dùng mã còn hạn, đổi mật khẩu,\\tokenVersion + 1, xóa refresh token}
  \ret{svc}{app}{200, hoặc 400 nếu mã sai hay hết hạn}
\end{seqdiagram}
